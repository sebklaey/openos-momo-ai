Schnelltest (Node 18+, keine Abhängigkeiten)

# 1. Artefakte ziehen
mkdir -p openos-audit && cd openos-audit
curl -fS https://openos.kimi.page/llms.txt -o llms.txt
curl -fS https://openos.kimi.page/verifikation/modell-selftest.mjs -o modell-selftest.mjs
curl -fS https://openos.kimi.page/kompression/assets/bitseed-engine.js -o bitseed-engine.js

# 2. Modell-Beweis (lädt Gewichte, prüft hart — exit 1 bei Fehlern)
node modell-selftest.mjs
